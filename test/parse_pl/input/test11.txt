sub expand_tabs {
    my ($text) = @_;
    my $tab_width_local = $tab_width // 8;

    my @lines = split /(\r?\n)/, $text;  # preserve line endings in split
    my $result = '';

    for (my $i = 0; $i < @lines; $i += 2) {
        my $line = $lines[$i];
        my $newline = $lines[$i + 1] // '';

        my $expanded_line = '';
        my $col = 0;
        foreach my $char (split //, $line) {
            if ($char eq "\t") {
                my $spaces = $tab_width_local - ($col % $tab_width_local);
                $expanded_line .= ' ' x $spaces;
                $col += $spaces;
            } else {
                $expanded_line .= $char;
                $col++;
            }
        }
        $result .= $expanded_line . $newline;
    }
    return $result;
}
